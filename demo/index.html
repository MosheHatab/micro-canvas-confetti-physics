Install

import { confetti } from 'micro-canvas-confetti-physics';

// Quick micro burst — ends in ~1s
confetti({ preset: 'spark' });

// Custom short burst with reach limit
confetti({ particleCount: 20, duration: 'short', scalar: 0.6, burstRadius: 200 });
